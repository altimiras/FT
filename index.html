async function loadData() { try { const res = await fetch('albumes.txt'); const text = await res.text(); // Procesa el texto para Android de forma limpia rawData = text .split('\n') .map(line => line.trim()) .filter(line => line.length > 0 && !line.startsWith('#')) .map(line => { // Separa por la barra vertical | const parts = line.split('|').map(item => item.trim()); return { anio: parts[0] || '0000', grupo: parts[1] || '1', titulo: parts[2] || 'Sin título', link: parts[3] || '#' }; }) .filter(item => item.link.startsWith('http')); // Asegura que solo tome registros con enlaces válidos populateYearSelect(); render(); } catch (err) { document.getElementById('counter').innerText = 'Error al cargar albumes.txt'; } }
